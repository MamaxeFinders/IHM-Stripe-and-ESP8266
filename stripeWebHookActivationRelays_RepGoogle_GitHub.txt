#include <ESP8266WiFi.h>
#include <ESP8266WebServer.h>
#include <ArduinoJson.h> //https://arduinojson.org/?utm_source=meta&utm_medium=library.properties

#define ARDUINOJSON_DEFAULT_NESTING_LIMIT 20
static StaticJsonDocument<2500> jsonDocument;

constexpr uint8_t RELAY_COUNT = 4;
constexpr unsigned long WIFI_RETRY_INTERVAL_MS = 30000;
constexpr unsigned long RELAY_ON_TIME_MS = 500;
constexpr unsigned long RELAY_COOLDOWN_MS = 500;

struct RelayCommand {
  const uint8_t* on;
  size_t onLength;
  const uint8_t* off;
  size_t offLength;
};

// DEFINE ALL RELAY CHANNEL COMMANDS
const uint8_t RELAY_1_ON[] = {0xA0, 0x01, 0x01, 0xA2};   // Hex command to send to serial for open relay
const uint8_t RELAY_1_OFF[] = {0xA0, 0x01, 0x00, 0xA1};  // Hex command to send to serial for close relay
const uint8_t RELAY_2_ON[] = {0xA0, 0x02, 0x01, 0xA3};   // Hex command to send to serial for open relay
const uint8_t RELAY_2_OFF[] = {0xA0, 0x02, 0x00, 0xA2};  // Hex command to send to serial for close relay
const uint8_t RELAY_3_ON[] = {0xA0, 0x03, 0x01, 0xA4};   // Hex command to send to serial for open relay
const uint8_t RELAY_3_OFF[] = {0xA0, 0x03, 0x00, 0xA3};  // Hex command to send to serial for close relay
const uint8_t RELAY_4_ON[] = {0xA0, 0x04, 0x01, 0xA5};   // Hex command to send to serial for open relay
const uint8_t RELAY_4_OFF[] = {0xA0, 0x04, 0x00, 0xA4};  // Hex command to send to serial for close relay

const RelayCommand RELAYS[RELAY_COUNT] = {
  {RELAY_1_ON, sizeof(RELAY_1_ON), RELAY_1_OFF, sizeof(RELAY_1_OFF)},
  {RELAY_2_ON, sizeof(RELAY_2_ON), RELAY_2_OFF, sizeof(RELAY_2_OFF)},
  {RELAY_3_ON, sizeof(RELAY_3_ON), RELAY_3_OFF, sizeof(RELAY_3_OFF)},
  {RELAY_4_ON, sizeof(RELAY_4_ON), RELAY_4_OFF, sizeof(RELAY_4_OFF)},
};

struct SessionRuntime {
  int8_t relayIndex = -1;
  bool relayIsOn = false;
  unsigned long nextActionAt = 0;
};

unsigned long wifiStatusMillis = 0;
uint16_t pendingSessions[RELAY_COUNT] = {0};
SessionRuntime runtime;

ESP8266WebServer server(80); // ROUTER PORT

// WIFI SETTINGS
const char* ssid = "XXXXXX";     // WiFi router
const char* password = "XXXXX";  // WiFi pwd

void sendRelayCommand(const RelayCommand& relay, bool turnOn) {
  if (turnOn) {
    Serial.write(relay.on, relay.onLength);
  } else {
    Serial.write(relay.off, relay.offLength);
  }
}

void allRelaysOff() {
  for (uint8_t relay = 0; relay < RELAY_COUNT; ++relay) {
    sendRelayCommand(RELAYS[relay], false);
    delay(5); // short guard time between commands
  }
}

void connectToWifi() {
  WiFi.mode(WIFI_STA);
  WiFi.setSleepMode(WIFI_NONE_SLEEP);
  WiFi.begin(ssid, password);

  Serial.print("Connecting to WiFi");
  uint32_t startAttempt = millis();
  while (WiFi.status() != WL_CONNECTED && millis() - startAttempt < 15000) {
    delay(250);
    Serial.print('.');
    yield();
  }
  Serial.println();

  if (WiFi.status() == WL_CONNECTED) {
    Serial.print("IP address: ");
    Serial.println(WiFi.localIP());
  } else {
    Serial.println("Failed to connect to WiFi");
  }
}

void setup() {

  Serial.begin(115200);
  connectToWifi();
  WiFi.setAutoReconnect(true);
  WiFi.persistent(true);

  server.on("/body", handleBody);
  server.begin(); // Start the server
  Serial.println("Server listening");

  allRelaysOff();
}

void ensureWifiConnection() {
  if (WiFi.status() == WL_CONNECTED) {
    return;
  }

  Serial.println("WiFi connection lost, attempting reconnection");
  WiFi.disconnect();
  connectToWifi();
}

void logWifiStatus(unsigned long currentMillis) {
  if (currentMillis - wifiStatusMillis < WIFI_RETRY_INTERVAL_MS) {
    return;
  }
  wifiStatusMillis = currentMillis;

  wl_status_t status = WiFi.status();
  switch (status) {
    case WL_NO_SSID_AVAIL:
      Serial.println("Configured SSID cannot be reached");
      break;
    case WL_CONNECTED:
      Serial.println("Connection successfully established");
      break;
    case WL_CONNECT_FAILED:
      Serial.println("Connection failed");
      break;
    default:
      Serial.printf("WiFi status code: %d\n", status);
      break;
  }
  if (status == WL_CONNECTED) {
    Serial.print("RSSI: ");
    Serial.println(WiFi.RSSI());
  }
}

void processPendingSessions() {
  unsigned long now = millis();

  if (runtime.relayIndex >= 0) {
    if (now < runtime.nextActionAt) {
      return;  // waiting for the next action window
    }

    uint8_t relay = static_cast<uint8_t>(runtime.relayIndex);
    const RelayCommand& command = RELAYS[relay];

    if (runtime.relayIsOn) {
      sendRelayCommand(command, false);
      runtime.relayIsOn = false;
      runtime.nextActionAt = now + RELAY_COOLDOWN_MS;

      if (pendingSessions[relay] > 0) {
        pendingSessions[relay]--;
      }

      if (pendingSessions[relay] == 0) {
        runtime.relayIndex = -1;
      }
    } else {
      if (pendingSessions[relay] == 0) {
        runtime.relayIndex = -1;
        return;
      }
      sendRelayCommand(command, true);
      runtime.relayIsOn = true;
      runtime.nextActionAt = now + RELAY_ON_TIME_MS;
    }
    return;
  }

  // No active relay, find the next one with pending sessions
  for (uint8_t relay = 0; relay < RELAY_COUNT; ++relay) {
    if (pendingSessions[relay] == 0) {
      continue;
    }

    runtime.relayIndex = relay;
    runtime.relayIsOn = false;
    runtime.nextActionAt = 0; // trigger immediate start
    break;
  }
}

void loop() { // Boucle avec les commandes

  unsigned long currentMillis = millis();
  ensureWifiConnection();
  logWifiStatus(currentMillis);

  server.handleClient(); // LISTEN THE NETWORK
  processPendingSessions();

  delay(1); // give time for background tasks
}

bool enqueueSessions(uint8_t relayNumber, uint16_t sessions) {
  if (relayNumber == 0 || relayNumber > RELAY_COUNT || sessions == 0) {
    return false;
  }

  pendingSessions[relayNumber - 1] += sessions;
  return true;
}

String describeRelayQueue() {
  String description;
  for (uint8_t i = 0; i < RELAY_COUNT; ++i) {
    if (i != 0) {
      description += F(", ");
    }
    description += F("R");
    description += (i + 1);
    description += F(":");
    description += pendingSessions[i];
  }
  return description;
}

void handleBody() { // Handler for the body path
  if (!server.hasArg("plain")) { // Check if body received
    server.send(400, "text/plain", "Requete recue sans donnees");
    return;
  }

  DeserializationError error = deserializeJson(jsonDocument, server.arg("plain"));
  if (error) {
    server.send(400, "text/plain", "JSON invalide: " + String(error.c_str()));
    return;
  }

  JsonVariant metadata = jsonDocument["data"]["object"]["metadata"];
  String piste = metadata["Piste"].as<String>();
  String prixUneSession = metadata["PrixUneSession"].as<String>();
  String prixDeuxSessions = metadata["PrixDeuxSessions"].as<String>();
  String credit = jsonDocument["data"]["object"]["amount_total"].as<String>();
  String transactionId = jsonDocument["id"].as<String>();

  uint8_t relayNumber = piste.toInt();
  uint16_t sessions = 0;

  if (credit == prixUneSession) {
    sessions = 1;
  } else if (credit == prixDeuxSessions) {
    sessions = 2;
  }

  if (!enqueueSessions(relayNumber, sessions)) {
    server.send(422, "text/plain", "Parametres de session invalides");
    return;
  }

  Serial.println("PISTE : " + piste);
  Serial.println("CREDIT : " + credit);
  Serial.println("PRIX UNE SESSION : " + prixUneSession);
  Serial.println("PRIX DEUX SESSIONS : " + prixDeuxSessions);
  Serial.println("NOMBRE DE SESSIONS EN FILE : " + describeRelayQueue());

  server.send(200, "text/plain",
              "Succes : =>" + transactionId + "<= Piste " + piste + " enfilee pour " + credit +
                  " euros (" + sessions + " sessions). Queue: " + describeRelayQueue());
}
